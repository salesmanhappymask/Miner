local function clamp(n, a, b)
  if n < a then return a end
  if n > b then return b end
  return n
end

local function fmtPct(n, digits)
  digits = digits == nil and 1 or digits
  local s = string.format("%." .. tostring(digits) .. "f%%", tonumber(n) or 0)
  s = s:gsub("%.0%%", "%%")
  return s
end

local function fmtTemp(n)
  local s = string.format("%.1f C", tonumber(n) or 0)
  s = s:gsub("%.0 C", " C")
  return s
end

local function formatRF(n)
  n = tonumber(n) or 0
  local a = math.abs(n)
  if a >= 1000000 then return string.format("%.2f MRF", n / 1000000) end
  if a >= 1000 then return string.format("%.1f KRF", n / 1000) end
  return string.format("%d RF", math.floor(n + 0.5))
end

local function formatRFPerTick(n)
  return formatRF(n) .. "/T"
end

local function formatAmount(n, suffix)
  n = tonumber(n) or 0
  local a = math.abs(n)
  local out
  if a >= 1000000 then
    out = string.format("%.2fM", n / 1000000)
  elseif a >= 1000 then
    out = string.format("%.1fK", n / 1000)
  else
    out = string.format("%d", math.floor(n + 0.5))
  end
  if suffix and suffix ~= "" then out = out .. " " .. suffix end
  return out
end

local function signedRF(n)
  n = tonumber(n) or 0
  if n > 0 then return "+" .. formatRF(n) end
  if n < 0 then return "-" .. formatRF(math.abs(n)) end
  return "0 RF"
end

local function signedPct(n)
  n = tonumber(n) or 0
  if math.abs(n) < 0.005 then return "0.00%" end
  return string.format("%+.2f%%", n)
end

local function roundNearest(n)
  n = tonumber(n) or 0
  if n >= 0 then return math.floor(n + 0.5) end
  return math.ceil(n - 0.5)
end

local function centerText(text, width)
  text = tostring(text or "")
  if width <= 0 then return "" end
  if #text > width then return text:sub(1, width) end
  local left = math.floor((width - #text) / 2)
  local right = width - #text - left
  return string.rep(" ", left) .. text .. string.rep(" ", right)
end

local function makeUI(m)
  local w, h = m.getSize()
  local cache = {}

  local function setBg(bg)
    if IS_COLOR then m.setBackgroundColor(bg) else m.setBackgroundColor(colors.black) end
  end

  local function setFg(fg)
    if IS_COLOR then m.setTextColor(fg) else m.setTextColor(colors.white) end
  end

  local function rect(x, y, ww, hh, bg)
    if ww <= 0 or hh <= 0 then return end
    local x1 = clamp(x, 1, w)
    local y1 = clamp(y, 1, h)
    local x2 = clamp(x + ww - 1, 1, w)
    local y2 = clamp(y + hh - 1, 1, h)
    if x2 < x1 or y2 < y1 then return end
    setBg(bg)
    local line = string.rep(" ", x2 - x1 + 1)
    for yy = y1, y2 do
      m.setCursorPos(x1, yy)
      m.write(line)
    end
  end

  local function field(key, x, y, ww, s, fg, bg)
    if ww <= 0 or y < 1 or y > h or x > w or x + ww - 1 < 1 then return end
    local x1 = clamp(x, 1, w)
    local x2 = clamp(x + ww - 1, 1, w)
    local visibleW = x2 - x1 + 1
    if visibleW <= 0 then return end
    s = tostring(s or "")
    if x < 1 then s = s:sub(2 - x) end
    if #s > visibleW then s = s:sub(1, visibleW) end
    local out = s .. string.rep(" ", visibleW - #s)
    local k1 = key .. "_out"
    local k2 = key .. "_fg"
    local k3 = key .. "_bg"
    if cache[k1] == out and cache[k2] == fg and cache[k3] == bg then return end
    setBg(bg or C_BG)
    setFg(fg or C_TEXT)
    m.setCursorPos(x1, y)
    m.write(out)
    cache[k1] = out
    cache[k2] = fg
    cache[k3] = bg
  end

  local function clearAll()
    setBg(C_BG)
    setFg(C_TEXT)
    m.clear()
    cache = {}
  end

  return { w = w, h = h, rect = rect, field = field, clearAll = clearAll }
end

local function safeCall(device, name, ...)
  if device == nil then return nil end
  local fn = device[name]
  if type(fn) ~= "function" then return nil end
  local ok, value = pcall(fn, ...)
  if not ok then return nil end
  return value
end

local function callOptional(name, ...)
  return safeCall(r, name, ...)
end

local function callEnergyValue(device, methods)
  if device == nil then return nil, nil end
  for _, method in ipairs(methods) do
    local fn = device[method]
    if type(fn) == "function" then
      local ok, value = pcall(fn, "RF")
      if ok and tonumber(value) ~= nil then return tonumber(value), method end
      ok, value = pcall(fn, "")
      if ok and tonumber(value) ~= nil then return tonumber(value), method end
      ok, value = pcall(fn)
      if ok and tonumber(value) ~= nil then return tonumber(value), method end
    end
  end
  return nil, nil
end

local function discoverEnergyCells()
  energyCells = {}
  energyCellNames = {}
  local names = peripheral.getNames and peripheral.getNames() or {}
  table.sort(names)
  for _, name in ipairs(names) do
    local ptype = peripheral.getType(name)
    if ptype == "cofh_thermalexpansion_energycell" or tostring(name):match("^cofh_thermalexpansion_energycell_") then
      local cell = peripheral.wrap(name)
      if cell then
        energyCells[#energyCells + 1] = cell
        energyCellNames[#energyCellNames + 1] = name
      end
    end
  end
end

local function getEnergyCellSnapshot(index)
  local cell = energyCells[index]
  if not cell then return nil end
  local stored, storedMethod = callEnergyValue(cell, { "getEnergyStored", "getEnergyLevel", "getEnergy" })
  local capacity, maxMethod = callEnergyValue(cell, { "getMaxEnergyStored", "getMaxEnergyLevel", "getEnergyCapacity", "getMaxEnergy" })
  local readable = stored ~= nil and capacity ~= nil and capacity > 0
  local pct = readable and clamp(stored / capacity * 100, 0, 100) or 0
  return {
    index = index,
    name = energyCellNames[index] or ("ENERGY CELL " .. tostring(index)),
    stored = stored or 0,
    capacity = capacity or 0,
    pct = pct,
    readable = readable,
    storedMethod = storedMethod,
    maxMethod = maxMethod,
  }
end

local function discoverTurbines()
  turbines = {}
  turbineNames = {}
  local names = peripheral.getNames and peripheral.getNames() or {}
  table.sort(names)
  for _, name in ipairs(names) do
    if peripheral.getType(name) == "BigReactors-Turbine" then
      local turbine = peripheral.wrap(name)
      if turbine then
        turbines[#turbines + 1] = turbine
        turbineNames[#turbineNames + 1] = name
        if turbineControl[name] == nil then
          turbineControl[name] = {
            lastRPM = nil,
            rpmRate = 0,
            rpmError = 0,
            rpmIntegral = 0,
            lastFlowStep = 0,
            flowEngaged = nil,
            flowDisengaged = nil,
            state = "WAITING",
          }
        end
      end
    end
  end
  if selectedTurbine < 1 then selectedTurbine = 1 end
  if #turbines > 0 and selectedTurbine > #turbines then selectedTurbine = #turbines end
  if #turbines == 0 then selectedTurbine = 1 end
end

local function getTurbineSnapshot(index)
  local t = turbines[index]
  if not t then return nil end
  local active = safeCall(t, "getActive")
  local rpm = safeCall(t, "getRotorSpeed") or 0
  local output = safeCall(t, "getEnergyProducedLastTick") or 0
  local stored = safeCall(t, "getEnergyStored") or 0
  local storedMax = safeCall(t, "getEnergyStoredMax") or 0
  local coil = safeCall(t, "getInductorEngaged")
  local flow = safeCall(t, "getFluidFlowRate") or 0
  local flowLimit = safeCall(t, "getFluidFlowRateMax") or 0
  local flowLimitMax = safeCall(t, "getFluidFlowRateMaxMax") or 0
  local inputAmount = safeCall(t, "getInputAmount")
  local outputAmount = safeCall(t, "getOutputAmount")
  local fluidMax = safeCall(t, "getFluidAmountMax")
  local storedPct = 0
  if storedMax and storedMax > 0 then storedPct = clamp(stored / storedMax * 100, 0, 100) end
  local flowPct = 0
  if flowLimitMax and flowLimitMax > 0 then flowPct = clamp(flowLimit / flowLimitMax * 100, 0, 100) end
  local name = turbineNames[index] or ("TURBINE " .. tostring(index))
  local control = turbineControl[name] or {}
  return {
    index = index,
    name = name,
    active = active == true,
    rpm = rpm,
    rpmError = targetRPM - rpm,
    rpmRate = control.rpmRate or 0,
    governorState = control.state or "WAITING",
    lastFlowStep = control.lastFlowStep or 0,
    output = output,
    stored = stored,
    storedMax = storedMax,
    storedPct = storedPct,
    coil = coil == true,
    flow = flow,
    flowLimit = flowLimit,
    flowLimitMax = flowLimitMax,
    flowPct = flowPct,
    inputAmount = inputAmount,
    outputAmount = outputAmount,
    fluidMax = fluidMax,
  }
end

local function getTurbineFleet()
  local fleet = {
    count = #turbines,
    active = 0,
    coils = 0,
    totalOutput = 0,
    totalStored = 0,
    totalStoredMax = 0,
    totalFlow = 0,
    totalFlowLimit = 0,
    totalFlowLimitMax = 0,
    rpmSum = 0,
    maxRPM = 0,
    minRPM = nil,
  }
  for i = 1, #turbines do
    local s = getTurbineSnapshot(i)
    if s then
      if s.active then
        fleet.active = fleet.active + 1
        fleet.rpmSum = fleet.rpmSum + (s.rpm or 0)
        fleet.totalFlow = fleet.totalFlow + (s.flow or 0)
        fleet.totalFlowLimit = fleet.totalFlowLimit + (s.flowLimit or 0)
        if fleet.minRPM == nil or (s.rpm or 0) < fleet.minRPM then fleet.minRPM = s.rpm or 0 end
        if (s.rpm or 0) > fleet.maxRPM then fleet.maxRPM = s.rpm or 0 end
      end
      if s.coil then fleet.coils = fleet.coils + 1 end
      fleet.totalOutput = fleet.totalOutput + (s.output or 0)
      fleet.totalStored = fleet.totalStored + (s.stored or 0)
      fleet.totalStoredMax = fleet.totalStoredMax + (s.storedMax or 0)
      fleet.totalFlowLimitMax = fleet.totalFlowLimitMax + (s.flowLimitMax or 0)
    end
  end
  fleet.avgRPM = fleet.active > 0 and fleet.rpmSum / fleet.active or 0
  fleet.minRPM = fleet.minRPM or 0
  fleet.storedPct = fleet.totalStoredMax > 0 and clamp(fleet.totalStored / fleet.totalStoredMax * 100, 0, 100) or 0
  return fleet
end
local function getEnergyBankSnapshot(activeCooled, fleet)
  fleet = fleet or getTurbineFleet()
  local reactorStored = safeCall(r, "getEnergyStored") or 0
  local reactorMax = safeCall(r, "getEnergyStoredMax")
  if reactorMax == nil or reactorMax <= 0 then reactorMax = maxEnergy end

  local bank = {
    reactorStored = reactorStored,
    reactorMax = reactorMax,
    reactorPct = reactorMax > 0 and clamp(reactorStored / reactorMax * 100, 0, 100) or 0,
    turbineStored = fleet.totalStored or 0,
    turbineMax = fleet.totalStoredMax or 0,
    cellsStored = 0,
    cellsMax = 0,
    cellsReadable = 0,
    cellsCount = #energyCells,
    cells = {},
    totalStored = 0,
    totalMax = 0,
  }

  for i = 1, #energyCells do
    local c = getEnergyCellSnapshot(i)
    bank.cells[#bank.cells + 1] = c
    if c and c.readable then
      bank.cellsStored = bank.cellsStored + c.stored
      bank.cellsMax = bank.cellsMax + c.capacity
      bank.cellsReadable = bank.cellsReadable + 1
    end
  end

  bank.cellsPct = bank.cellsMax > 0 and clamp(bank.cellsStored / bank.cellsMax * 100, 0, 100) or 0
  bank.turbinePct = bank.turbineMax > 0 and clamp(bank.turbineStored / bank.turbineMax * 100, 0, 100) or 0
  bank.totalStored = bank.reactorStored + bank.turbineStored + bank.cellsStored
  bank.totalMax = bank.reactorMax + bank.turbineMax + bank.cellsMax
  bank.totalPct = bank.totalMax > 0 and clamp(bank.totalStored / bank.totalMax * 100, 0, 100) or 0
  return bank
end

  if ww < 4 or hh < 3 then return end
  ui.rect(x, y, ww, hh, C_BG)
  ui.rect(x, y, ww, 1, C_DIV)
  ui.rect(x, y + hh - 1, ww, 1, C_DIV)

  ui.rect(x + ww - 1, y, 1, hh, C_DIV)
  ui.rect(x + 1, y + 1, ww - 2, hh - 2, C_INSET)
  ui.rect(x + 1, y + 1, 1, hh - 2, accentColor or C_ACCENT)
end

local function drawTab(key, x, y, ww, text, active, action)
  local bg = active and C_BRAND or C_DIV
  ui.rect(x, y, ww, 2, bg)
  ui.field(key, x, y + 1, ww, centerText(string.upper(text), ww), C_TEXT, bg)
  addHotspot(x, y, ww, 2, action)

local function kv(key, x, y, width, label, value, valueColor)
  if width <= 0 then return end
  local labelText = string.upper(tostring(label or "")) .. ":"
  local valueText = tostring(value or "")
  local labelW = math.min(#labelText, width)
  ui.field(key .. "_l", x, y, labelW, labelText, C_MUTED, C_INSET)
  local avail = width - labelW - 1
  if avail > 0 then
    ui.field(key .. "_v", x + labelW + 1, y, avail, valueText, valueColor or C_TEXT, C_INSET)
  end
end

local function drawBar(x, y, width, percent, fillColor)
  if width <= 0 then return end
  percent = clamp(percent or 0, 0, 100)
  local filled = clamp(math.floor(width * percent / 100 + 0.5), 0, width)
  mon.setCursorPos(x, y)
  mon.setBackgroundColor(fillColor)
  if filled > 0 then mon.write(string.rep(" ", filled)) end
  mon.setBackgroundColor(C_DIV)
  if width - filled > 0 then mon.write(string.rep(" ", width - filled)) end
  mon.setBackgroundColor(C_BG)
end

local function drawButton(key, x, y, ww, hh, text, bg, fg, action)
  if ww <= 0 or hh <= 0 then return end
  local face = bg or C_BRAND
  ui.rect(x, y, ww, hh, C_DIV)
  if ww > 2 and hh > 2 then
    ui.rect(x + 1, y + 1, ww - 2, hh - 2, face)
    ui.field(key, x + 1, y + math.floor((hh - 2) / 2) + 1, ww - 2, centerText(string.upper(text), ww - 2), fg or C_TEXT, face)
  else
    ui.rect(x, y, ww, hh, face)
    ui.field(key, x, y, ww, centerText(string.upper(text), ww), fg or C_TEXT, face)
  end
  if action then addHotspot(x, y, ww, hh, action) end
end

local function percentColor(p)
  if p < 30 then return C_BAD end
  if p < 70 then return C_WARN end
  return C_OK
end

local function splitTwo(x, ww, gap)
  gap = gap or 2
  local leftW = math.floor((ww - gap) / 2)
  local rightW = ww - gap - leftW
  local rightX = x + leftW + gap
  return leftW, rightX, rightW
end

local function getSnapshot()
  local temp = r.getFuelTemperature()
  local fuel = r.getFuelAmount()
  local fuelMax = r.getFuelAmountMax()
  local energy = r.getEnergyStored()
  local gen = r.getEnergyProducedLastTick()
  local active = r.getActive()
  local rod = r.getControlRodLevel(0)
  local casingTemp = callOptional("getCasingTemperature")
  local reactivity = callOptional("getFuelReactivity")
  local fuelUse = callOptional("getFuelConsumedLastTick")
  local waste = callOptional("getWasteAmount")
  local activeCooled = callOptional("isActivelyCooled") == true
  local hotFluidProduced = callOptional("getHotFluidProducedLastTick")
  local hotFluidAmount = callOptional("getHotFluidAmount")
  local hotFluidMax = callOptional("getHotFluidAmountMax")
  local coolantAmount = callOptional("getCoolantAmount")
  local coolantMax = callOptional("getCoolantAmountMax")
  local fleet = getTurbineFleet()
  local fuelPct = fuelMax > 0 and clamp(fuel / fuelMax * 100, 0, 100) or 0
  local reactorEnergyPct = clamp(energy / maxEnergy * 100, 0, 100)
  local hotFluidPct = 0
  if hotFluidAmount ~= nil and hotFluidMax ~= nil and hotFluidMax > 0 then
    hotFluidPct = clamp(hotFluidAmount / hotFluidMax * 100, 0, 100)
  end
  local coolantPct = 0
  if coolantAmount ~= nil and coolantMax ~= nil and coolantMax > 0 then
    coolantPct = clamp(coolantAmount / coolantMax * 100, 0, 100)
  end
  local energyBank = getEnergyBankSnapshot(activeCooled, fleet)
  local controlStored = energyBank.totalStored
  local controlMax = energyBank.totalMax
  local controlSource = "GRID RESERVE"
  local controlPct = energyBank.totalPct
  local deviation = controlPct - targetPercent
  return {
    temp = temp,
    fuel = fuel,
    fuelMax = fuelMax,
    fuelPct = fuelPct,
    energy = energy,
    energyPct = reactorEnergyPct,
    reactorEnergyPct = reactorEnergyPct,
    gen = gen,
    active = active,
    rod = rod,
    deviation = deviation,
    casingTemp = casingTemp,
    reactivity = reactivity,
    fuelUse = fuelUse,
    waste = waste,
    activeCooled = activeCooled,
    hotFluidProduced = hotFluidProduced,
    hotFluidAmount = hotFluidAmount,
    hotFluidMax = hotFluidMax,
    hotFluidPct = hotFluidPct,
    coolantAmount = coolantAmount,
    coolantMax = coolantMax,
    coolantPct = coolantPct,
    fleet = fleet,
    energyBank = energyBank,
    controlStored = controlStored,
    controlMax = controlMax,
    controlPct = controlPct,
    controlSource = controlSource,
    steamDemand = lastSteamDemand,
    steamProduced = lastSteamProduced,
    steamBalance = lastSteamBalance,
    steamBufferPct = lastSteamBufferPct,
    plantState = plantControlState,
    targetRPM = targetRPM,
    stateText = active and "ACTIVE" or "SHUTDOWN",
    stateColor = active and C_OK or C_BAD,
    modeText = autoEnabled and "AUTOMATIC" or "MANUAL",
  }
end

local function drawStatCard(key, x, y, ww, hh, title, accent, items)
  cardSurface(x, y, ww, hh, accent)
  sectionTitle(key .. "_t", x, y, ww, title)
  local startY = y + 2
  local usableRows = hh - 3
  for i = 1, math.min(#items, usableRows) do
    local item = items[i]
    kv(key .. i, x + 3, startY + i - 1, ww - 5, item[1], item[2], item[3])
  end
end

local function drawMeterCard(key, x, y, ww, hh, title, label, detail, pct, fillColor)
  cardSurface(x, y, ww, hh, C_ACCENT)
  sectionTitle(key .. "_t", x, y, ww, title)
  local innerW = ww - 5
  local leftW = math.floor(innerW * 0.42)
  ui.field(key .. "_label", x + 3, y + 2, leftW, string.upper(label), C_MUTED, C_INSET)
  ui.field(key .. "_detail", x + 3 + leftW, y + 2, innerW - leftW, detail, C_TEXT, C_INSET)
  drawBar(x + 2, y + 3, ww - 4, pct, fillColor)
end

local function drawTabs()
  local labels = { "OVERVIEW", "REACTOR CTRL", "TURBINES", "TURBINE CTRL", "ENERGY" }
  local pages = { "overview", "manual", "turbines", "turbine_manual", "energy" }
  local actions = { "tab_overview", "tab_manual", "tab_turbines", "tab_turbine_manual", "tab_energy" }
  local gap = 1
  local totalW = W - 2
  local tabW = math.floor((totalW - gap * 4) / 5)
  local x = 2
  for i = 1, 5 do
    local width = i == 5 and (W - x + 1) or tabW
    drawTab("tab_" .. tostring(i), x, 3, width, labels[i], currentPage == pages[i], actions[i])
    x = x + width + gap
  end
end
local function drawOverview(d)
  local outerX = 2
  local outerW = W - 2
  local contentTop = 6
  local contentBottom = H - 1
  local gap = 1
  local topH = 7
  local meterH = 5
  local teleY = contentTop + topH + gap + meterH + gap + meterH + gap + meterH + gap
  local teleH = contentBottom - teleY + 1

local function drawButtonRow(prefix, x, y, ww, labels, actions, bg)
  local count = #labels
  if count == 0 then return end
  local gap = 1
  local btnW = math.floor((ww - gap * (count - 1)) / count)
  local cursorX = x
  for i = 1, count do
    local width = (i == count) and (x + ww - cursorX) or btnW
    local face = bg
    if type(bg) == "table" then face = bg[i] end
    drawButton(prefix .. i, cursorX, y, width, 3, labels[i], face or C_BRAND, C_TEXT, actions[i])
    cursorX = cursorX + width + gap
  end
end

local function getControlEnergy()
  local activeCooled = callOptional("isActivelyCooled") == true
  local fleet = getTurbineFleet()

  return bank.totalStored, bank.totalMax
end

local function runPassiveEnergyControl()
  local stored, capacity = getControlEnergy()
  local percent = capacity > 0 and clamp(stored / capacity * 100, 0, 100) or 0

  if previousPercent == nil then
    previousPercent = percent
    lastControlStep = 0
    lastControlRate = 0
    plantControlState = "ENERGY HOLD"
    return
  end

  local errorAmt = percent - targetPercent
  local rate = percent - previousPercent
  previousPercent = percent
  lastControlRate = rate
  local adjustment = kp * errorAmt + kd * rate
  local step = roundNearest(adjustment)
  local cap = normalCap
  if percent < dangerThreshold and rate < 0 then cap = dangerCap end
  local current = r.getControlRodLevel(0)
  if step > 0 and current >= sensitiveZoneStart then
    local roomLeft = 100 - current
    local scaledCap = math.max(1, math.floor(roomLeft / 4))
    if scaledCap < cap then cap = scaledCap end
  end
  step = clamp(step, -cap, cap)
  lastControlStep = step
  if step ~= 0 then
    r.setAllControlRodLevels(clamp(current + step, 0, 100))
  end
  if math.abs(errorAmt) <= 2 then
    plantControlState = "ENERGY BALANCED"
  elseif errorAmt < 0 then
    plantControlState = "RAISING OUTPUT"
  else
    plantControlState = "TRIMMING OUTPUT"
  end
end

local function runTurbineGovernor()
  local totalDemand = 0
  local bank = getEnergyBankSnapshot(true, getTurbineFleet())
  local gridPct = bank.totalPct
  local highBuffer = clamp(targetPercent + 3, 5, 98)
  local lowBuffer = clamp(targetPercent - 3, 2, 95)
  local spinupThreshold = targetRPM - 100
  local overspeedThreshold = targetRPM + 100
  local criticalThreshold = targetRPM + 200

  for i = 1, #turbines do
    local t = getTurbineSnapshot(i)
    local device = turbines[i]
    local control = turbineControl[turbineNames[i]]
    if control == nil then
      control = { lastRPM = nil, rpmRate = 0, rpmError = 0, rpmIntegral = 0, lastFlowStep = 0, flowEngaged = nil, flowDisengaged = nil, state = "WAITING" }
      turbineControl[turbineNames[i]] = control
    end

    if t and t.active then
      local rpmRate = 0
      if control.lastRPM ~= nil then rpmRate = t.rpm - control.lastRPM end
      control.lastRPM = t.rpm
      control.rpmRate = rpmRate
      control.rpmError = targetRPM - t.rpm

      if turbineGovernorEnabled then
        local currentFlow = t.flowLimit or 0
        local maximum = t.flowLimitMax or 0
        if maximum <= 0 then maximum = 2000 end
        local desiredCoil = t.coil
        local newFlow = currentFlow

        if t.rpm >= criticalThreshold then
          desiredCoil = true
          newFlow = 0
          control.rpmIntegral = 0
          control.state = "OVERSPEED CUT"
        else
          if t.rpm < spinupThreshold then
            desiredCoil = false
          elseif t.rpm > overspeedThreshold then
            desiredCoil = true
          elseif t.storedPct >= 98 then
            desiredCoil = false
          elseif gridPct >= highBuffer then
            desiredCoil = false
          elseif gridPct <= lowBuffer and t.rpm >= targetRPM - 50 then
            desiredCoil = true
          end

          local rpmError = targetRPM - t.rpm
          control.rpmIntegral = clamp((control.rpmIntegral or 0) + rpmError, -rpmIntegralLimit, rpmIntegralLimit)
          local rawStep = rpmKp * rpmError + rpmKi * control.rpmIntegral - rpmKd * rpmRate
          local absError = math.abs(rpmError)
          local stepCap = rpmBaseFlowStep
          if absError > 400 then
            stepCap = rpmMaxFlowStep
          elseif absError > 150 then
            stepCap = 250
          elseif absError > 50 then
            stepCap = 100
          end
          if absError <= rpmDeadband then rawStep = rpmKi * control.rpmIntegral - rpmKd * rpmRate end
          if t.rpm > overspeedThreshold then rawStep = math.min(rawStep, -100) end

          local baseFlow = currentFlow
          if desiredCoil ~= t.coil then
            if desiredCoil and control.flowEngaged ~= nil then
              baseFlow = control.flowEngaged
            elseif not desiredCoil and control.flowDisengaged ~= nil then
              baseFlow = control.flowDisengaged
            elseif not desiredCoil then
              baseFlow = math.max(0, currentFlow - 200)
            end
          end

          local flowStep = roundNearest(clamp(rawStep, -stepCap, stepCap))
          newFlow = clamp(baseFlow + flowStep, 0, maximum)

          if t.rpm < spinupThreshold then
            control.state = "SPINNING UP"
          elseif (gridPct >= highBuffer or t.storedPct >= 98) and not desiredCoil then
            control.state = "GRID HOLD"
          elseif t.rpm > overspeedThreshold then
            control.state = "BRAKING"
          elseif math.abs(rpmError) <= rpmDeadband then
            control.state = "RPM LOCKED"
          elseif rpmError > 0 then
            control.state = "ACCELERATING"
          else
            control.state = "DECELERATING"
          end
        end

        if desiredCoil ~= t.coil then safeCall(device, "setInductorEngaged", desiredCoil) end
        local actualStep = roundNearest(newFlow - currentFlow)
        if math.abs(actualStep) >= 1 then safeCall(device, "setFluidFlowRateMax", newFlow) end
        control.lastFlowStep = actualStep
        if desiredCoil then
          control.flowEngaged = newFlow
        else
          control.flowDisengaged = newFlow
        end
        totalDemand = totalDemand + newFlow
      else
        control.rpmIntegral = 0
        if t.rpm >= criticalThreshold then
          safeCall(device, "setInductorEngaged", true)
          safeCall(device, "setFluidFlowRateMax", 0)
          control.state = "SAFETY OVERSPEED CUT"
          control.lastFlowStep = -(t.flowLimit or 0)
        else
          control.state = "MANUAL FLOW"
          control.lastFlowStep = 0
          totalDemand = totalDemand + (t.flowLimit or 0)
        end
      end
    else
      if control then
        control.lastRPM = nil
        control.rpmRate = 0
        control.rpmError = 0
        control.rpmIntegral = 0
        control.lastFlowStep = 0
        control.state = "OFFLINE"
      end
    end
  end

  return totalDemand
end

local function runSteamReactorControl(steamDemand, controlRods)
  local produced = callOptional("getHotFluidProducedLastTick") or 0
  local hotAmount = callOptional("getHotFluidAmount")
  local hotMax = callOptional("getHotFluidAmountMax")
  local coolantAmount = callOptional("getCoolantAmount")
  local coolantMax = callOptional("getCoolantAmountMax")
  local hotPct = nil
  local coolantPct = 100
  if hotAmount ~= nil and hotMax ~= nil and hotMax > 0 then hotPct = clamp(hotAmount / hotMax * 100, 0, 100) end
  if coolantAmount ~= nil and coolantMax ~= nil and coolantMax > 0 then coolantPct = clamp(coolantAmount / coolantMax * 100, 0, 100) end

  lastSteamDemand = steamDemand
  lastSteamProduced = produced
  lastSteamBalance = produced - steamDemand
  lastSteamBufferPct = hotPct or 0

  local current = r.getControlRodLevel(0)

  if controlRods == false then
    lastControlStep = 0
    lastControlRate = 0
    previousSteamError = nil
    steamIntegral = 0
    if turbineGovernorEnabled then
      plantControlState = "RPM AUTO / REACTOR MANUAL"
    else
      plantControlState = "MANUAL"
    end
    return
  end

  if coolantPct < 5 then
    local step = math.min(dangerCap, 100 - current)
    lastControlStep = step
    lastControlRate = 0
    if step > 0 then r.setAllControlRodLevels(current + step) end
    plantControlState = "LOW COOLANT HOLD"
    previousSteamError = nil
    steamIntegral = 0
    return
  end

  if steamDemand <= 1 then
    local step = math.min(normalCap, 100 - current)
    lastControlStep = step
    lastControlRate = 0
    if step > 0 then r.setAllControlRodLevels(current + step) end
    plantControlState = "STEAM IDLE"
    previousSteamError = nil
    steamIntegral = 0
    return
  end

  local denom = math.max(steamDemand, produced, 100)
  local productionError = clamp((produced - steamDemand) / denom * 100, -100, 100)
  local derivative = 0
  if previousSteamError ~= nil then derivative = productionError - previousSteamError end
  previousSteamError = productionError
  steamIntegral = clamp(steamIntegral + productionError, -steamIntegralLimit, steamIntegralLimit)
  local bufferError = hotPct ~= nil and (hotPct - steamBufferTargetPercent) or 0
  local adjustment = steamProductionKp * productionError + steamProductionKi * steamIntegral + steamBufferKp * bufferError + steamProductionKd * derivative
  local step = roundNearest(adjustment)
  local cap = normalCap
  if produced < steamDemand * 0.75 and (hotPct == nil or hotPct < 25) then cap = dangerCap end
  if step > 0 and current >= sensitiveZoneStart then
    local roomLeft = 100 - current
    local scaledCap = math.max(1, math.floor(roomLeft / 4))
    if scaledCap < cap then cap = scaledCap end
  end
  step = clamp(step, -cap, cap)
  lastControlStep = step
  lastControlRate = derivative
  if step ~= 0 then r.setAllControlRodLevels(clamp(current + step, 0, 100)) end

  local tolerance = math.max(50, steamDemand * 0.05)
  if lastSteamBalance < -tolerance then
    plantControlState = "RAISING STEAM"
  elseif lastSteamBalance > tolerance then
    plantControlState = "TRIMMING STEAM"
  else
    plantControlState = "STEAM BALANCED"
  end
end

local function autoControl()
  if callOptional("isActivelyCooled") == true then
    local steamDemand = runTurbineGovernor()
    runSteamReactorControl(steamDemand, autoEnabled)
    previousPercent = nil
  else
    if autoEnabled then
      runPassiveEnergyControl()
    else
      previousPercent = nil
      lastControlStep = 0
      lastControlRate = 0
      plantControlState = "MANUAL"
    end
    previousSteamError = nil
    lastSteamDemand = 0
    lastSteamProduced = 0
    lastSteamBalance = 0
    lastSteamBufferPct = 0
  end
end

local function updateStorageDelta()
  local energy = getControlEnergy()
  if lastEnergySample == nil then
    storageDelta = 0
  else
    storageDelta = energy - lastEnergySample
  end
  lastEnergySample = energy
end

local function primePlantTelemetry()
  if callOptional("isActivelyCooled") ~= true then return end
  local fleet = getTurbineFleet()
  local produced = callOptional("getHotFluidProducedLastTick") or 0
  local hotAmount = callOptional("getHotFluidAmount")
  local hotMax = callOptional("getHotFluidAmountMax")
  lastSteamDemand = fleet.totalFlowLimit or 0
  lastSteamProduced = produced
  lastSteamBalance = produced - lastSteamDemand
  if hotAmount ~= nil and hotMax ~= nil and hotMax > 0 then
    lastSteamBufferPct = clamp(hotAmount / hotMax * 100, 0, 100)
  else
    lastSteamBufferPct = 0
  end
end

local function applyAction(action)
  if action == "tab_overview" then
    currentPage = "overview"
    return
  elseif action == "tab_manual" then
    currentPage = "manual"
    return
  elseif action == "tab_turbines" then
    currentPage = "turbines"
    return
  elseif action == "tab_turbine_manual" then
    currentPage = "turbine_manual"
    return
  elseif action == "tab_energy" then
    currentPage = "energy"
    return
  elseif action == "start" then
    r.setActive(true)
    return
  elseif action == "stop" then
    r.setActive(false)
    return
  elseif action == "toggle_auto" then
    autoEnabled = not autoEnabled
    previousPercent = nil
    previousSteamError = nil
    steamIntegral = 0
    lastControlStep = 0
    lastControlRate = 0
    return
  elseif action == "turbine_rescan" then
    discoverTurbines()
    discoverEnergyCells()
    previousPercent = nil
    return
  elseif action == "turbine_prev" then
    if #turbines > 0 then
      selectedTurbine = selectedTurbine - 1
      if selectedTurbine < 1 then selectedTurbine = #turbines end
    end
    return
  elseif action == "turbine_next" then
    if #turbines > 0 then
      selectedTurbine = selectedTurbine + 1
      if selectedTurbine > #turbines then selectedTurbine = 1 end
    end
    return
  elseif action == "toggle_turbine_governor" then
    turbineGovernorEnabled = not turbineGovernorEnabled
    for name, control in pairs(turbineControl) do
      control.lastRPM = nil
      control.rpmRate = 0
      control.rpmIntegral = 0
      control.lastFlowStep = 0
      control.state = turbineGovernorEnabled and "WAITING" or "MANUAL FLOW"
    end
    return
  end

  local rodDelta = action:match("^rod_([%+%-]?%d+)$")
  if rodDelta then
    if autoEnabled then return end
    local current = r.getControlRodLevel(0)
    r.setAllControlRodLevels(clamp(current + (tonumber(rodDelta) or 0), 0, 100))
    return
  end

  local rodSet = action:match("^rodset_(%d+)$")
  if rodSet then
    if autoEnabled then return end
    r.setAllControlRodLevels(clamp(tonumber(rodSet) or 0, 0, 100))
    return
  end

  local targetDelta = action:match("^target_([%+%-]?%d+)$")
  if targetDelta then
    targetPercent = clamp(targetPercent + (tonumber(targetDelta) or 0), 0, 100)
    previousPercent = nil
    return
  end

  local targetSet = action:match("^targetset_(%d+)$")
  if targetSet then
    targetPercent = clamp(tonumber(targetSet) or targetPercent, 0, 100)
    previousPercent = nil
    return
  end

  local steamTargetDelta = action:match("^steamtarget_([%+%-]?%d+)$")
  if steamTargetDelta then
    steamBufferTargetPercent = clamp(steamBufferTargetPercent + (tonumber(steamTargetDelta) or 0), 10, 90)
    previousSteamError = nil
    return
  end

  local rpmDelta = action:match("^rpm_([%+%-]?%d+)$")
  if rpmDelta then
    targetRPM = clamp(targetRPM + (tonumber(rpmDelta) or 0), 100, 3000)
    for name, control in pairs(turbineControl) do
      control.lastRPM = nil
      control.rpmIntegral = 0
    end
    return
  end

  local rpmSet = action:match("^rpmset_(%d+)$")
  if rpmSet then
    targetRPM = clamp(tonumber(rpmSet) or targetRPM, 100, 3000)
    for name, control in pairs(turbineControl) do
      control.lastRPM = nil
      control.rpmIntegral = 0
    end
    return
  end

  local intervalSet = action:match("^interval_set(%d+)$")
  if intervalSet then
    checkInterval = clamp(tonumber(intervalSet) or checkInterval, 1, 30)
    return
  end

  local intervalDelta = action:match("^interval_delta_([%+%-]?%d+)$")
  if intervalDelta then
    checkInterval = clamp(checkInterval + (tonumber(intervalDelta) or 0), 1, 30)
    return
  end

  local t = turbines[selectedTurbine]
  if action == "turbine_start" then
    if t then safeCall(t, "setActive", true) end
    return
  elseif action == "turbine_stop" then
    if t then safeCall(t, "setActive", false) end
    return
  elseif action == "turbine_coil_on" then
    if t and not turbineGovernorEnabled then safeCall(t, "setInductorEngaged", true) end
    return
  elseif action == "turbine_coil_off" then
    if t and not turbineGovernorEnabled then safeCall(t, "setInductorEngaged", false) end
    return
  end

  local flowDelta = action:match("^tflow_([%+%-]?%d+)$")
  if flowDelta and t then
    if turbineGovernorEnabled then return end
    local current = safeCall(t, "getFluidFlowRateMax") or 0
    local maximum = safeCall(t, "getFluidFlowRateMaxMax") or 0
    if maximum <= 0 then maximum = 2000 end
    safeCall(t, "setFluidFlowRateMax", clamp(current + (tonumber(flowDelta) or 0), 0, maximum))
    return
  end

  local flowPreset = action:match("^tflowset_(%d+)$")
  if flowPreset and t then
    if turbineGovernorEnabled then return end
    local maximum = safeCall(t, "getFluidFlowRateMaxMax") or 0
    if maximum <= 0 then maximum = 2000 end
    local value = math.floor(maximum * clamp(tonumber(flowPreset) or 0, 0, 100) / 100 + 0.5)
    safeCall(t, "setFluidFlowRateMax", value)
    return
  end

  if action == "turbines_start_all" or action == "turbines_stop_all" or action == "turbines_coils_on" or action == "turbines_coils_off" then
    if turbineGovernorEnabled and (action == "turbines_coils_on" or action == "turbines_coils_off") then return end
    for i = 1, #turbines do
      if action == "turbines_start_all" then safeCall(turbines[i], "setActive", true) end
      if action == "turbines_stop_all" then safeCall(turbines[i], "setActive", false) end
      if action == "turbines_coils_on" then safeCall(turbines[i], "setInductorEngaged", true) end
      if action == "turbines_coils_off" then safeCall(turbines[i], "setInductorEngaged", false) end
    end
    return
  end
end
local function hotspotActionAt(x, y)
  for i = 1, #hotspots do
    local h = hotspots[i]
    if x >= h.x1 and x <= h.x2 and y >= h.y1 and y <= h.y2 then
      return h.action
    end
  end
  return nil
end

discoverTurbines()
discoverEnergyCells()
primePlantTelemetry()
updateStorageDelta()
draw()
timerId = os.startTimer(checkInterval)

local ok, err = pcall(function()
  while true do
    local event, a, b, c = os.pullEvent()
    if event == "monitor_touch" then
      local action = hotspotActionAt(b, c)
      if action then
        applyAction(action)
        draw()
      end
    elseif event == "timer" and a == timerId then
      autoControl()
      updateStorageDelta()
      draw()
      timerId = os.startTimer(checkInterval)
    end
  end
end)

if not ok then
  term.setBackgroundColor(colors.black)
  term.setTextColor(colors.white)
  term.clear()
  term.setCursorPos(1, 1)
  print("Program stopped: " .. tostring(err))
end

local function wrapThis(name)
  local w, i = nil, 0
  while w == nil and i <= 100 do
    w = peripheral.wrap(name .. "_" .. i)
    i = i + 1
  end
  return w
end

local function supportsColor(m)
  return m.isColor and m.isColor() or false
end

local IS_COLOR = supportsColor(mon)

local C_HEADER = IS_COLOR and colors.lightGray or colors.black

local C_INSET = colors.black

local C_MUTED = IS_COLOR and colors.lightGray or colors.white

local function clamp(n, a, b)
  if n < a then return a end
  if n > b then return b end
  return n
end

local function makeUI(m)
  local w, h = m.getSize()
  local cache = {}

  local function setBg(bg)
    if IS_COLOR then m.setBackgroundColor(bg) else m.setBackgroundColor(colors.black) end
  end

  local function setFg(fg)
    if IS_COLOR then m.setTextColor(fg) else m.setTextColor(colors.white) end
  end

  local function rect(x, y, ww, hh, bg)
    if ww <= 0 or hh <= 0 then return end
    local x1 = clamp(x, 1, w)
    local y1 = clamp(y, 1, h)
    local x2 = clamp(x + ww - 1, 1, w)
    local y2 = clamp(y + hh - 1, 1, h)
    if x2 < x1 or y2 < y1 then return end
    setBg(bg)
    local line = string.rep(" ", x2 - x1 + 1)
    for yy = y1, y2 do
      m.setCursorPos(x1, yy)
      m.write(line)
    end
  end

  local function field(key, x, y, ww, s, fg, bg)
    if ww <= 0 or y < 1 or y > h or x > w or x + ww - 1 < 1 then return end
    local x1 = clamp(x, 1, w)
    local x2 = clamp(x + ww - 1, 1, w)
    local visibleW = x2 - x1 + 1
    if visibleW <= 0 then return end
    s = tostring(s or "")
    if x < 1 then s = s:sub(2 - x) end
    if #s > visibleW then s = s:sub(1, visibleW) end
    local out = s .. string.rep(" ", visibleW - #s)
    local k1 = key .. "_out"
    local k2 = key .. "_fg"
    local k3 = key .. "_bg"
    if cache[k1] == out and cache[k2] == fg and cache[k3] == bg then return end
    setBg(bg or C_BG)
    setFg(fg or C_TEXT)
    m.setCursorPos(x1, y)
    m.write(out)
    cache[k1] = out
    cache[k2] = fg
    cache[k3] = bg
  end

  local function clearAll()
    setBg(C_BG)
    setFg(C_TEXT)
    m.clear()
    cache = {}
  end

  return { w = w, h = h, rect = rect, field = field, clearAll = clearAll }
end

local ui = makeUI(mon)

local function callOptional(name)
  local fn = r[name]
  if type(fn) ~= "function" then return nil end
  local ok, value = pcall(fn)
  if not ok then return nil end
  return value
end

local function percentColor(p)
  if p < 30 then return C_BAD end
  if p < 70 then return C_WARN end
  return C_OK
end

local function formatRF(n)
  n = tonumber(n) or 0
  local a = math.abs(n)
  if a >= 1000000 then return string.format("%.2f MRF", n / 1000000) end
  if a >= 1000 then return string.format("%.1f kRF", n / 1000) end
  return string.format("%d RF", math.floor(n + 0.5))
end

local function formatAmount(n, suffix)
  n = tonumber(n) or 0
  local a = math.abs(n)
  local out
  if a >= 1000000 then
    out = string.format("%.2fM", n / 1000000)
  elseif a >= 1000 then
    out = string.format("%.1fk", n / 1000)
  else
    out = string.format("%d", math.floor(n + 0.5))
  end
  if suffix and suffix ~= "" then out = out .. " " .. suffix end
  return out
end

local function signedRF(n)
  n = tonumber(n) or 0
  if n > 0 then return "+" .. formatRF(n) end
  if n < 0 then return "-" .. formatRF(math.abs(n)) end
  return "0 RF"
end

local function signedPercent(n)
  n = tonumber(n) or 0
  if math.abs(n) < 0.005 then return "0.00%" end
  return string.format("%+.2f%%", n)
end

local function centerText(text, width)
  text = tostring(text or "")
  if width <= 0 then return "" end
  if #text > width then return text:sub(1, width) end
  local left = math.floor((width - #text) / 2)
  local right = width - #text - left
  return string.rep(" ", left) .. text .. string.rep(" ", right)
end

local function cardSurface(x, y, ww, hh, accentColor)
  if ww < 4 or hh < 3 then return end
  ui.rect(x, y, ww, hh, C_BG)
  ui.rect(x, y, ww, 1, C_DIV)
  ui.rect(x, y + hh - 1, ww, 1, C_DIV)
  ui.rect(x, y, 1, hh, C_DIV)
  ui.rect(x + ww - 1, y, 1, hh, C_DIV)
  ui.rect(x + 1, y + 1, ww - 2, hh - 2, C_INSET)
  ui.rect(x + 1, y + 1, 1, hh - 2, accentColor or C_ACCENT)
end

local function sectionTitle(key, x, y, ww, title, color)
  ui.field(key, x + 3, y, math.max(0, ww - 5), title, color or C_TEXT, C_DIV)
end

local function drawHeader(title)
  ui.rect(1, 1, W, 1, C_HEADER)
  ui.rect(1, 2, W, 1, C_DIV)
  local brand = "ZelOS"
  local tx = 1 + math.floor((W - #title) / 2)
  ui.field("hdr_title", tx, 1, #title, title, C_TEXT, C_HEADER)
  ui.field("hdr_brand", W - #brand + 1, 1, #brand, brand, C_BRAND, C_HEADER)
end

local function clearHotspots()
  hotspots = {}
end

local function addHotspot(x, y, ww, hh, action)
  hotspots[#hotspots + 1] = { x1 = x, y1 = y, x2 = x + ww - 1, y2 = y + hh - 1, action = action }
end

local function drawButton(key, x, y, ww, hh, text, bg, fg, action)
  if ww <= 0 or hh <= 0 then return end
  ui.rect(x, y, ww, hh, C_DIV)
  if ww > 2 and hh > 2 then
    ui.rect(x + 1, y + 1, ww - 2, hh - 2, bg)
    ui.field(key, x + 1, y + math.floor(hh / 2), ww - 2, centerText(text, ww - 2), fg or C_TEXT, bg)
  else
    ui.rect(x, y, ww, hh, bg)
    ui.field(key, x, y + math.floor(hh / 2), ww, centerText(text, ww), fg or C_TEXT, bg)
  end
  if action then addHotspot(x, y, ww, hh, action) end
end

local function drawTab(key, x, y, ww, text, active, action)
  local bg = active and C_BRAND or C_DIV
  local fg = C_TEXT
  ui.field(key, x, y, ww, centerText(text, ww), fg, bg)
  addHotspot(x, y, ww, 1, action)
end

local function drawBar(x, y, width, percent, fillColor)
  if width <= 0 then return end
  percent = clamp(percent or 0, 0, 100)
  local filled = math.floor(width * percent / 100 + 0.5)
  filled = clamp(filled, 0, width)
  mon.setCursorPos(x, y)
  mon.setBackgroundColor(fillColor)
  if filled > 0 then mon.write(string.rep(" ", filled)) end
  mon.setBackgroundColor(C_DIV)
  if width - filled > 0 then mon.write(string.rep(" ", width - filled)) end
  mon.setBackgroundColor(C_BG)
end

local function kv(key, x, y, width, label, value, valueColor)
  if width <= 0 then return end
  local labelText = tostring(label or "") .. ":"
  local valueText = tostring(value or "")
  local labelW = math.min(#labelText, width)
  ui.field(key .. "_l", x, y, labelW, labelText, C_MUTED, C_INSET)
  local avail = width - labelW - 1
  if avail > 0 then
    ui.field(key .. "_v", x + labelW + 1, y, avail, valueText, valueColor or C_TEXT, C_INSET)
  end
end

local function drawStatCard(key, x, y, ww, hh, title, accent, items)
  cardSurface(x, y, ww, hh, accent)
  sectionTitle(key .. "_t", x, y, ww, title, C_TEXT)
  local rows = hh - 2
  for i = 1, math.min(#items, rows) do
    local item = items[i]
    kv(key .. "_kv" .. i, x + 3, y + i, ww - 5, item[1], item[2], item[3])
  end
end

local function drawMeterCard(key, x, y, ww, hh, title, accent, label, detail, pct, fillColor)
  cardSurface(x, y, ww, hh, accent)
  sectionTitle(key .. "_t", x, y, ww, title, C_TEXT)
  ui.field(key .. "_label", x + 3, y + 1, math.max(1, math.floor((ww - 6) / 2)), label, C_MUTED, C_INSET)
  ui.field(key .. "_detail", x + 3 + math.floor((ww - 6) / 2), y + 1, math.max(1, ww - 6 - math.floor((ww - 6) / 2)), detail, C_TEXT, C_INSET)
  drawBar(x + 2, y + 2, ww - 4, pct, fillColor)
end

local function splitTwo(x, ww, gap)
  gap = gap or 2
  local leftW = math.floor((ww - gap) / 2)
  local rightW = ww - gap - leftW
  local rightX = x + leftW + gap
  return leftW, rightX, rightW
end

local function getReactorSnapshot()
  local temp = r.getFuelTemperature()
  local fuel = r.getFuelAmount()
  local fuelMax = r.getFuelAmountMax()
  local energy = r.getEnergyStored()
  local gen = r.getEnergyProducedLastTick()
  local active = r.getActive()
  local rod = r.getControlRodLevel(0)
  local casingTemp = callOptional("getCasingTemperature")
  local reactivity = callOptional("getFuelReactivity")
  local fuelUse = callOptional("getFuelConsumedLastTick")
  local waste = callOptional("getWasteAmount")

  local fuelPct = 0
  if fuelMax and fuelMax > 0 then fuelPct = clamp(fuel / fuelMax * 100, 0, 100) end
  local energyPct = clamp(energy / maxEnergy * 100, 0, 100)
  local deviation = energyPct - targetPercent

  return {
    temp = temp,
    fuel = fuel,
    fuelMax = fuelMax,
    fuelPct = fuelPct,
    energy = energy,
    energyPct = energyPct,
    gen = gen,
    active = active,
    rod = rod,
    deviation = deviation,
    casingTemp = casingTemp,
    reactivity = reactivity,
    fuelUse = fuelUse,
    waste = waste,
    modeText = autoEnabled and "AUTOMATIC" or "MANUAL",
    stateText = active and "ACTIVE" or "SHUTDOWN",
    stateColor = active and C_OK or C_BAD,
  }
end

local function getDefaultStatus()
  if currentPage == "overview" then
    return "TAB-UI 2.1 | Overview. Touch MANUAL & CONFIG for controls and tuning.", C_MUTED
  end
  if autoEnabled then
    return "Automatic mode is ON. Manual rod controls are locked until AUTO is turned OFF.", C_WARN
  end
  return "Automatic mode is OFF. Manual rod controls now change insertion directly.", C_OK
end

local function setStatus(msg, color)
  statusMessage = tostring(msg or "")
  statusColor = color or C_MUTED
end

local function drawTabs()
  local leftW = 18

  local rightW = 24
  if leftW + gap + rightW > W - 2 then

    rightW = W - 2 - gap - leftW

  drawTab("tab1", 2, 3, leftW, "OVERVIEW", currentPage == "overview", "tab_overview")
  drawTab("tab2", 2 + leftW + gap, 3, rightW, "MANUAL & CONFIG", currentPage == "manual", "tab_manual")
end

end

local function drawDeltaButtons(prefix, baseAction, x, y, ww)
  local labels = { "-10", "-5", "-1", "+1", "+5", "+10" }
  local values = { -10, -5, -1, 1, 5, 10 }
  local gap = 1
  local btnW = math.floor((ww - gap * 5) / 6)
  local cursorX = x
  for i = 1, 6 do
    local width = (i == 6) and (x + ww - cursorX) or btnW
    local action = baseAction .. tostring(values[i])
    drawButton(prefix .. i, cursorX, y, width, 3, labels[i], C_DIV, C_TEXT, action)
    cursorX = cursorX + width + gap
  end
end

local function draw()
  clearHotspots()
  ui.clearAll()
  drawHeader("Reactor Control v2.1")
  drawTabs()
  local d = getReactorSnapshot()

  if W >= 50 then
    ui.field("hdr_state", 2, 1, 11, d.stateText, d.stateColor, C_HEADER)
  end

  if currentPage == "manual" then
    drawManualPage(d)
  else
    drawOverviewPage(d)
  end

  drawFooter()
  sleep(0)
end

local function autoControl()
  if not autoEnabled then
    previousPercent = nil
    lastControlStep = 0
    lastControlRate = 0
    return
  end

  local energy = r.getEnergyStored()
  local percent = clamp(energy / maxEnergy * 100, 0, 100)

  if previousPercent == nil then
    previousPercent = percent
    lastControlStep = 0
    lastControlRate = 0
    return
  end

  local errorAmt = percent - targetPercent
  local rate = percent - previousPercent
  previousPercent = percent
  lastControlRate = rate

  local adjustment = kp * errorAmt + kd * rate
  local step = math.floor(adjustment + 0.5)
  local cap = normalCap

  if percent < dangerThreshold and rate < 0 then cap = dangerCap end

  local current = r.getControlRodLevel(0)
  if step > 0 and current >= sensitiveZoneStart then
    local roomLeft = 100 - current
    local scaledCap = math.max(1, math.floor(roomLeft / 4))
    if scaledCap < cap then cap = scaledCap end
  end

  if step > cap then step = cap end
  if step < -cap then step = -cap end

  lastControlStep = step
  if step ~= 0 then
    local newLevel = clamp(current + step, 0, 100)
    r.setAllControlRodLevels(newLevel)
  end
end

local function updateStorageDelta()
  local energy = r.getEnergyStored()
  if lastEnergySample == nil then
    storageDelta = 0
  else
    storageDelta = energy - lastEnergySample
  end
  lastEnergySample = energy
end

local function applyAction(action)
  if action == "tab_overview" then
    currentPage = "overview"
    setStatus("Overview page selected.", C_MUTED)
    return
  elseif action == "tab_manual" then
    currentPage = "manual"
    setStatus("Manual & configuration page selected.", C_MUTED)
    return
  elseif action == "start" then
    r.setActive(true)
    setStatus("Reactor started.", C_OK)
    return
  elseif action == "stop" then
    r.setActive(false)
    setStatus("Reactor shut down.", C_BAD)
    return
  elseif action == "toggle_auto" then
    autoEnabled = not autoEnabled
    previousPercent = nil
    lastControlStep = 0
    lastControlRate = 0
    if autoEnabled then
      setStatus("Automatic control enabled.", C_OK)
    else
      setStatus("Automatic control disabled. Manual rod controls unlocked.", C_WARN)
    end
    return
  end

  local rodDelta = action:match("^rod_([%+%-]?%d+)$")
  if rodDelta then
    if autoEnabled then
      setStatus("Disable AUTO before adjusting control rods manually.", C_WARN)
      return
    end
    local delta = tonumber(rodDelta) or 0
    local current = r.getControlRodLevel(0)
    local newLevel = clamp(current + delta, 0, 100)
    r.setAllControlRodLevels(newLevel)
    setStatus(string.format("Control rods set to %d%% inserted.", math.floor(newLevel + 0.5)), C_OK)
    return
  end

  local targetDelta = action:match("^target_([%+%-]?%d+)$")
  if targetDelta then
    local delta = tonumber(targetDelta) or 0
    targetPercent = clamp(targetPercent + delta, 0, 100)
    previousPercent = nil
    setStatus(string.format("Stored-energy target set to %d%%.", targetPercent), C_OK)
    return
  end

  local intervalSet = action:match("^interval_set(%d+)$")
  if intervalSet then
    checkInterval = clamp(tonumber(intervalSet) or checkInterval, 1, 30)
    setStatus(string.format("Check interval set to %d second(s).", checkInterval), C_OK)
    return
  end

  local intervalDelta = action:match("^interval_([%+%-]?%d+)$")
  if intervalDelta then
    checkInterval = clamp(checkInterval + (tonumber(intervalDelta) or 0), 1, 30)
    setStatus(string.format("Check interval set to %d second(s).", checkInterval), C_OK)
    return
  end
end

local function hotspotActionAt(x, y)
  for i = 1, #hotspots do
    local h = hotspots[i]
    if x >= h.x1 and x <= h.x2 and y >= h.y1 and y <= h.y2 then
      return h.action
    end
  end
  return nil
end

if W < 58 or H < 28 then
  error("Monitor is too small for the tabbed reactor interface at text scale 0.5")
end

updateStorageDelta()
draw()
timerId = os.startTimer(checkInterval)

local ok, err = pcall(function()
  while true do
    local event, a, b, c = os.pullEvent()

    if event == "monitor_touch" then
      local action = hotspotActionAt(b, c)
      if action then
        applyAction(action)
        draw()
      end
    elseif event == "timer" and a == timerId then
      autoControl()
      updateStorageDelta()
      draw()
      timerId = os.startTimer(checkInterval)
    end
  end
end)

if not ok then
  term.setBackgroundColor(colors.black)
  term.setTextColor(colors.white)
  term.clear()
  term.setCursorPos(1, 1)
  print("Program stopped: " .. tostring(err))
end
